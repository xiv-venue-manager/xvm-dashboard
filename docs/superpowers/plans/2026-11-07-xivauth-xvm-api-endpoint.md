# XIV Auth Verified Characters: xvm-api Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** xvm-api records that a person owns FFXIV characters proven by XIV Auth, and a verified claim takes a character from an unverified holder but never from a verified one.

**Architecture:** One internal route, `POST /internal/characters/verified`, callable only with the dashboard's service credential. It calls a new `CharacterService.record_verified`, which locks the people involved in id order and applies four rules per character (created, verified, transferred, conflict). `person_characters` gains `xivauth_key` and two partial unique indexes. `CharacterRow` exposes `verified_via` so clients can show the badge.

**Tech Stack:** Python 3.14, FastAPI, SQLAlchemy 2 async, Alembic, pytest (asyncio_mode=auto), ruff, mypy. Repo: `xiv-venue-manager/xvm-api`, branch from `dev`.

Spec: `docs/superpowers/specs/2026-11-07-xivauth-character-verification-design.md` in `xvm-dashboard`. This plan is the xvm-api half. The dashboard half is `2026-11-07-xivauth-dashboard-flow.md` and depends on this one being merged.

Add inside `CharacterService`, after `release`:

```python
    async def record_verified(self, person_id: int, items: Sequence[VerifiedCharacter]) -> list[VerifyOutcome]:
        """What XIV Auth says a person owns. A verified claim moves a character off an unverified
        holder and never off a verified one. Everyone involved is locked in id order first."""
        ordered = sorted(items, key=lambda item: item.lodestone_id)
        await self._lock_people(person_id, ordered)
        return [await self._record_one(person_id, item) for item in ordered]

    async def _lock_people(self, person_id: int, items: Sequence[VerifiedCharacter]) -> None:
        locked: set[int] = set()
        for _ in range(3):
            wanted = {person_id}
            for item in items:
                row = await self._match(item)
                if row is not None:
                    wanted.add(row.person_id)
            fresh = sorted(wanted - locked)
            if not fresh:
                return
            for holder in fresh:
                await self._lock_person(holder)
            locked.update(fresh)

    async def _record_one(self, person_id: int, item: VerifiedCharacter) -> VerifyOutcome:
        row = await self._match(item)
        if row is None:
            created = PersonCharacterModel(person_id=person_id, is_primary=not await self._holds_any(person_id))
            self._prove(created, item)
            self.session.add(created)
            await self.session.flush()
            return self._outcome(item, "created", created.id)
        if await self._name_belongs_to_another(row, item):
            return self._outcome(item, "conflict", None, _NAME_TAKEN)
        if row.person_id == person_id:
            self._prove(row, item)
            await self.session.flush()
            return self._outcome(item, "verified", row.id)
        if row.verified_via in _PROVEN:
            return self._outcome(item, "conflict", None, _ALREADY_PROVEN)
        displaced, was_primary = row.person_id, row.is_primary
        row.is_primary = not await self._holds_any(person_id)
        row.person_id = person_id
        self._prove(row, item)
        await self.session.flush()
        if was_primary:
            await self._promote_heir(displaced)
        await self._tell_displaced(displaced, row)
        logger.info(
            "character {} {} moved from person {} to person {} by an XIV Auth proof",
            row.character_name, row.world, displaced, person_id,
        )
        return self._outcome(item, "transferred", row.id)

    async def _match(self, item: VerifiedCharacter) -> PersonCharacterModel | None:
        by_lodestone = (
            await self.session.execute(select(PersonCharacterModel).where(PersonCharacterModel.lodestone_id == item.lodestone_id))
        ).scalar_one_or_none()
        if by_lodestone is not None:
            return by_lodestone
        return await self._find(item.character_name, item.world)

    async def _name_belongs_to_another(self, row: PersonCharacterModel, item: VerifiedCharacter) -> bool:
        named = await self._find(item.character_name, item.world)
        return named is not None and named.id != row.id

    @staticmethod
    def _prove(row: PersonCharacterModel, item: VerifiedCharacter) -> None:
        row.character_name = item.character_name
        row.world = item.world
        row.lodestone_id = item.lodestone_id
        row.xivauth_key = item.xivauth_key
        row.verified_via = "xivauth"
        row.verified_at = _now()

    @staticmethod
    def _outcome(item: VerifiedCharacter, outcome: Literal["created", "verified", "transferred", "conflict"], character_id: int | None, detail: str | None = None) -> VerifyOutcome:
        return VerifyOutcome(item.lodestone_id, item.character_name, item.world, outcome, character_id, detail)

    async def _tell_displaced(self, person_id: int, row: PersonCharacterModel) -> None:
        from api.services.platform import NotificationService  # platform imports this module's neighbours

        await NotificationService(self.session).notify(
            person_id,
            kind=NotificationKind.CharacterReassigned,
            title="A character link moved",
            body=f"{row.character_name} on {row.world} was verified through XIV Auth by its owner, so it is no longer linked to your account.",
            target_type="character",
            target_id=row.id,
        )
```

async def record_verified_characters(
    session: SessionDep, principal: PrincipalDep, payload: VerifiedCharactersIn
) -> list[VerifiedCharacterOutcome]:
    """Characters XIV Auth says the person owns, as the dashboard relays them. A person's own
    credential is refused, or anyone could mark a character verified."""
    _dashboard_only(principal)
    if await session.get(PersonModel, payload.person_id) is None:
        raise HTTPException(404, "No such person.")
    outcomes = await CharacterService(session).record_verified(
        payload.person_id,
        [VerifiedCharacter(c.lodestone_id, c.xivauth_key, c.character_name, c.world) for c in payload.characters],
    )
    return [
        VerifiedCharacterOutcome(
            character_name=o.character_name, world=o.world, lodestone_id=o.lodestone_id,
            outcome=o.outcome, character_id=o.character_id, detail=o.detail,
        )
        for o in outcomes
    ]
